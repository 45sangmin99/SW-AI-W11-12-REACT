cmake_minimum_required(VERSION 3.16)
project(mini_redis LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SANITIZE "Enable ASan and UBSan" OFF)
find_package(Threads REQUIRED)
add_compile_options(-Wall -Wextra -Wpedantic)
if(SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_executable(mini-redis src/main.cpp)
target_link_libraries(mini-redis PRIVATE Threads::Threads)
add_executable(unit-tests tests/unit.cpp)
target_link_libraries(unit-tests PRIVATE Threads::Threads)
enable_testing()
add_test(NAME unit COMMAND unit-tests)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:mini-redis>)
add_executable(engine-bench benchmarks/engine.cpp)
set_tests_properties(unit integration PROPERTIES TIMEOUT 120)
